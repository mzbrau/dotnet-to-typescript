  ExecuteScript:
import fs from "node:fs";
import path from "node:path";
import vm from "node:vm";

/**
 * Executes a standalone script in a fresh vm context with the current mocks injected.
 * Uses the same mock object references so mockReturnValue / spies apply.
 *
 * @param {string} scriptPath
 * @param {Record<string, unknown>} mockGlobals
 * @returns {Record<string, unknown>}
 */
export function executeScript(scriptPath, mockGlobals) {
  const absolutePath = path.resolve(scriptPath);
  const code = fs.readFileSync(absolutePath, "utf8");

  const context = {
    ...mockGlobals,
    console,
    globalThis: undefined,
  };

  // Allow scripts to see the injected globals as both bare identifiers and on globalThis.
  context.globalThis = context;

  vm.createContext(context);
  vm.runInContext(code, context, { filename: absolutePath });
  return context;

/**
 * Factory helpers for building model objects in tests.
 * Overrides are deep-merged onto generated defaults.
 */

function deepMerge(defaults, overrides) {
  if (overrides === undefined || overrides === null) return defaults;
  if (Array.isArray(defaults) || Array.isArray(overrides)) return overrides;
  if (typeof defaults !== "object" || defaults === null || typeof overrides !== "object") {
    return overrides;
  }
  const result = { ...defaults };
  for (const [key, value] of Object.entries(overrides)) {
    result[key] = key in defaults ? deepMerge(defaults[key], value) : value;
  }
  return result;
}

export function createCar(overrides = {}) {
  return deepMerge({ authorizedDrivers: [], fuelLevel: 0, isRunning: false, make: "", model: "", serviceCosts: {}, type: 0, year: 0 }, overrides);
}

export function createMaintenanceRecord(overrides = {}) {
  return deepMerge({ apiKey: "", cost: 0, description: "", isEnabled: false, isWarrantyWork: false, maxRetries: 0, mechanic: "", partsReplaced: [], serviceDate: new Date(0) }, overrides);
}

export function createUser(overrides = {}) {
  return deepMerge({ address: "", dateOfBirth: new Date(0), name: "", permissions: {}, phoneNumbers: [], status: 0 }, overrides);
}

import path from "node:path";
import { fileURLToPath } from "node:url";

const __dirname = path.dirname(fileURLToPath(import.meta.url));
const sampleScript = path.join(__dirname, "..", "scripts", "sample-script.js");

describe("generated mock environment", () => {
  test("configures mocks, executes a script, and verifies calls", () => {
    globalThis.dave.addPhoneNumber.mockReturnValue(undefined);

    globalThis.executeScript(sampleScript);

    expect(globalThis.dave.addPhoneNumber).toHaveBeenCalled();
  });

  test("throws when a method is called without mock configuration", () => {
    expect(() => globalThis.dave.addPhoneNumber()).toThrow(/No mock behaviour has been configured/);
  });

  test("resets property values between tests", () => {
    const original = globalThis.dave.address;
    globalThis.dave.address = "changed-in-test";
    globalThis.resetMocks();
    expect(globalThis.dave.address).toEqual(original);
  });

  test("builds model objects with factories", () => {
    const model = globalThis.createUser({ });
    expect(model).toBeTypeOf("object");
  });
});

# Generated Vitest environment

This folder was generated by `dotnet-to-typescript` with `--test`.

## Quick start

```bash
npm install
npm test
```

Coverage:

```bash
npm run test:coverage
```

## Write a test

Scripts are standalone (no imports). Mocks are installed as globals.

```javascript
dave.addPhoneNumber.mockReturnValue(/* ... */);

executeScript("path/to/your-script.js");

expect(dave.addPhoneNumber).toHaveBeenCalled();
```

Add `// @ts-check` at the top of scripts to type-check against `sample.d.ts`.

## More documentation

See the project docs for Vitest setup details, mock configuration, factories, and recommended layout:
https://github.com/mzbrau/dotnet-to-typescript
